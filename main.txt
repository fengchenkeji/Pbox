static
std::shared_ptr<spdlog::logger>
logger = nullptr;

bool is_debug(int argc, char* argv[]) {
    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "--run_type=debug") == 0)
            return true;
    }
    return false;
}


void call_so(const char* so_path, const char* so_function, bool debug) {
    // ✅ 创建 logger（如果不存在的话）
    if (!logger) {
        logger = spdlog::stdout_color_mt("console");
        spdlog::register_logger(logger);
    }

    if (debug) {
        SPDLOG_LOGGER_INFO(logger, "Loading SO: {}", so_path);
    }

    void* handle = dlopen(so_path, RTLD_LAZY);
    if (!handle) {
        DBG(logger, fmt::format("Cannot open file: {}", so_path));
        return;
    }

    dlerror();
    void* func = dlsym(handle, so_function);
    if (!func) {
        DBG(logger, fmt::format("Cannot find function: {} from {}", so_function, so_path));
    } else {
        ((void(*)())func)();
    }
    dlclose(handle);
}#include <string>

std::vector<std::string> get_link(const std::string filepath, const std::string exe, bool debug) {
    std::vector<std::string> link;
    // 修正路径拼接
    std::string fullFile = exe + "/res/images.txt";
    std::ifstream in(fullFile);

    // 获取全局logger
    auto logger = spdlog::get("console");
    if (!logger) {
        // 兜底：临时创建简易日志器，避免完全无日志
        logger = spdlog::stdout_color_mt("console");
        spdlog::register_logger(logger);
        spdlog::warn("Global console logger not initialized, auto create temporary logger");
    }

    if (debug) {
        if (in.is_open()) {
            SPDLOG_LOGGER_INFO(logger, "Loading file: {}", fullFile);
        } else {
            SPDLOG_LOGGER_WARN(logger, "File open failed: {}", fullFile);
        }
    }

    // 示例读取逻辑（你原有读取逻辑自行补全）
    std::string line;
    while (std::getline(in, line)) {
        if (!line.empty()) {
            link.push_back(line);
        }
    }
    in.close();

    return link;
}
